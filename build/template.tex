% 代码块
% pandoc 在未启用 --listings 时输出的是 verbatim 环境，而 verbatim 默认不折行，
% 长命令行会大幅溢出页面（实测最大超宽 508pt ≈ 18cm）。这里用 fvextra 的
% Verbatim 重定义 verbatim，恢复自动折行，并补上边框与底色，
% 使外观与原 --listings 方案（basicstyle/frame/backgroundcolor）保持一致。
\usepackage{fvextra}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{
  breaklines=true,
  breakanywhere=true,
  breaksymbolleft={},
  fontsize=\small,
  frame=single,
  framesep=3mm,
  rulecolor=\color{black!20},
  fillcolor=\color{black!3}
}

\lstset{
  basicstyle=\ttfamily\small,
  breaklines=true,
  columns=fullflexible,
  frame=single,
  rulecolor=\color{black!20},
  backgroundcolor=\color{black!3}
}
